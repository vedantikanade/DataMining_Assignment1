/**
 * Represents a single expense entry.
 */
class Expense {
    /**
     * @param {string} id - Unique identifier (UUID v4 style)
     * @param {number} amount - Expense amount in dollars
     * @param {string} description - Short description
     * @param {string} category - Category name
     * @param {string} date - ISO date string (YYYY-MM-DD)
     */
    constructor(id, amount, description, category, date) {
        this.id = id;
        this.amount = amount;
        this.description = description;
        this.category = category;
        this.date = date;
    }
}

/**
 * Handles persistence (localStorage) and business logic.
 */
class ExpenseTracker {
    constructor(storageKey = 'expenses') {
        this.storageKey = storageKey;
        this.expenses = this._loadFromStorage();
    }

    /** Load expenses from localStorage, returns [] if none */
    _loadFromStorage() {
        const raw = localStorage.getItem(this.storageKey);
        if (!raw) return [];
        try {
            const parsed = JSON.parse(raw);
            // Re‑instantiate as Expense objects for type safety
            return parsed.map(e => new Expense(e.id, e.amount, e.description, e.category, e.date));
        } catch (e) {
            console.error('Failed to parse stored expenses:', e);
            return [];
        }
    }

    /** Save current expense list to localStorage */
    _saveToStorage() {
        localStorage.setItem(this.storageKey, JSON.stringify(this.expenses));
    }

    /** Generate a simple UUID (RFC4122 version 4) */
    static _generateId() {
        // Not cryptographically strong but sufficient for client‑side IDs
        return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, c => {
            const r = Math.random() * 16 | 0;
            const v = c === 'x' ? r : (r & 0x3 | 0x8);
            return v.toString(16);
        });
    }

    /** Add a new expense and persist */
    addExpense({ amount, description, category, date }) {
        const expense = new Expense(
            ExpenseTracker._generateId(),
            parseFloat(amount),
            description.trim(),
            category.trim(),
            date
        );
        this.expenses.push(expense);
        this._saveToStorage();
        return expense;
    }

    /** Delete an expense by its id */
    deleteExpense(id) {
        this.expenses = this.expenses.filter(e => e.id !== id);
        this._saveToStorage();
    }

    /** Return a shallow copy of all expenses */
    getAllExpenses() {
        // Return a copy to avoid external mutation
        return [...this.expenses];
    }

    /** Filter expenses by category (case‑insensitive) */
    filterByCategory(category) {
        if (!category || category.toLowerCase() === 'all') {
            return this.getAllExpenses();
        }
        const lower = category.toLowerCase();
        return this.expenses.filter(e => e.category.toLowerCase() === lower);
    }

    /** Compute total amount of a given expense list */
    static computeTotal(expenseList) {
        return expenseList.reduce((sum, e) => sum + e.amount, 0);
    }

    /** Return a Set of unique categories (for filter dropdown) */
    getUniqueCategories() {
        const set = new Set();
        this.expenses.forEach(e => set.add(e.category));
        return set;
    }
}
